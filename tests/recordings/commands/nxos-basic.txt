# Cisco NX-OS (R2, lab devices only), with --os nxos.
# Log in as a network-admin. Changes the running config: creates and removes loopback999.
terminal length 0
show version
show running-config
show interface brief
show ip route
show inventory
show nonexistent
show i
show interface ethernet
show interface ethernet 99/99
configure terminal
interface loopback999
description Exscript test, safe to remove
end
show running-config interface loopback999
configure terminal
no interface loopback999
end
show this-command-is-much-longer-than-the-terminal-width | include something-that-does-not-match-anything-at-all
exit
